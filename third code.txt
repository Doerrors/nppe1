# 2. Paths & Class Names
BASE_PATH = "/kaggle/input/competitions/26-t-1-dl-gen-ainppe-1/"
IMG_DIR = os.path.join(BASE_PATH, "images")
train_df = pd.read_csv(os.path.join(BASE_PATH, "train.csv"))
class_names = train_df.columns[1:].tolist()

# 3. Stable Class Weights
counts = train_df[class_names].sum().values
weights = 1.0 / np.log1p(counts) 
weights = weights / weights.sum() * 20 
class_weights = torch.tensor(weights, dtype=torch.float).to('cuda')

# 4. Dataset Class
class ChestDataset(Dataset):
    def __init__(self, df, img_dir, transform=None, is_test=False):
        self.df, self.img_dir, self.transform, self.is_test = df, img_dir, transform, is_test
    def __len__(self): return len(self.df)
    def __getitem__(self, idx):
        name = self.df.iloc[idx, 0]
        path = os.path.join(self.img_dir, name if name.endswith('.png') else f"{name}.png")
        img = Image.open(path).convert('RGB')
        if self.transform: img = self.transform(img)
        if self.is_test: return img, torch.tensor(0), name
        label = torch.tensor(np.argmax(self.df.iloc[idx, 1:].values), dtype=torch.long)
        return img, label, name

# 5. Model, Loss, Optimizer
model = models.densenet121(weights='IMAGENET1K_V1')
model.classifier = nn.Linear(model.classifier.in_features, 20)
model = model.to('cuda')
criterion = nn.CrossEntropyLoss(weight=class_weights, label_smoothing=0.1)
optimizer = optim.AdamW(model.parameters(), lr=config.lr)

# --- CLOUD RESUME LOGIC ---
start_epoch = 0
try:
    artifact = run.use_artifact('densenet-checkpoint:latest')
    artifact_dir = artifact.download()
    checkpoint = torch.load(os.path.join(artifact_dir, 'last_model.pth'))
    model.load_state_dict(checkpoint['model_state_dict'])
    optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
    start_epoch = checkpoint['epoch']
    print(f"Resuming from Cloud at Epoch {start_epoch}")
except:
    print("Starting fresh run.")

# 6. Data Loaders
train_data, val_data = train_test_split(train_df, test_size=0.12, random_state=42)
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
])

train_loader = DataLoader(ChestDataset(train_data, IMG_DIR, transform), batch_size=4, shuffle=True, num_workers=2)
val_loader = DataLoader(ChestDataset(val_data, IMG_DIR, transform), batch_size=4, shuffle=False, num_workers=2)

accumulation_steps = 8 

# --- TRAINING LOOP ---
for epoch in range(start_epoch, config.epochs):
    model.train()
    t_loss = 0
    optimizer.zero_grad()
    
    for i, (imgs, labels, _) in enumerate(tqdm(train_loader, desc=f"Training Epoch {epoch+1}")):
        imgs, labels = imgs.to('cuda'), labels.to('cuda')
        outputs = model(imgs)
        loss = criterion(outputs, labels) / accumulation_steps
        loss.backward()
        
        if (i + 1) % accumulation_steps == 0:
            optimizer.step()
            optimizer.zero_grad()
        t_loss += loss.item() * accumulation_steps
    
    # --- VALIDATION & CONFUSION MATRIX ---
    model.eval()
    val_preds, val_true = [], []
    with torch.no_grad():
        for imgs, labels, _ in tqdm(val_loader, desc="Validating"):
            outputs = model(imgs.to('cuda'))
            preds = torch.argmax(outputs, dim=1).cpu().numpy()
            val_preds.extend(preds)
            val_true.extend(labels.numpy())

    # Log Confusion Matrix to W&B
    wandb.log({
        "conf_mat": wandb.plot.confusion_matrix(
            probs=None,
            y_true=val_true, 
            preds=val_preds,
            class_names=class_names
        ),
        "epoch": epoch + 1,
        "train_loss": t_loss / len(train_loader)
    })

    # Save Cloud Backup
    checkpoint = {'epoch': epoch+1, 'model_state_dict': model.state_dict(), 'optimizer_state_dict': optimizer.state_dict()}
    torch.save(checkpoint, 'last_model.pth')
    artifact = wandb.Artifact('densenet-checkpoint', type='model')
    artifact.add_file('last_model.pth')
    run.log_artifact(artifact)
    
    gc.collect()
    torch.cuda.empty_cache()

# 7. Final Submission
model.eval()
results = []
test_df = pd.read_csv(os.path.join(BASE_PATH, "test.csv"))
test_loader = DataLoader(ChestDataset(test_df, IMG_DIR, transform, is_test=True), batch_size=4)

with torch.no_grad():
    for imgs, _, ids in tqdm(test_loader, desc="Predicting Test Set"):
        preds = torch.argmax(model(imgs.to('cuda')), dim=1).cpu().numpy()
        for i, p in enumerate(preds):
            row = [0]*20; row[p] = 1
            results.append([ids[i]] + row)

pd.DataFrame(results, columns=['id']+class_names).to_csv("submission.csv", index=False)
