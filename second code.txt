# 2. Paths
BASE_PATH = "/kaggle/input/competitions/26-t-1-dl-gen-ainppe-1/"
IMG_DIR = os.path.join(BASE_PATH, "images")
train_df = pd.read_csv(os.path.join(BASE_PATH, "train.csv"))
# Get the 20 pathology classes [cite: 7, 100]
class_names = train_df.columns[1:].tolist()

# 3. Stable Weights Calculation [cite: 45, 153, 155]
# Using log1p smoothing to prevent weights from exploding and causing high loss
counts = train_df[class_names].sum().values
weights = 1.0 / np.log1p(counts) 
weights = weights / weights.sum() * 20 
class_weights = torch.tensor(weights, dtype=torch.float).to('cuda')

# 4. Corrected Dataset Class
class ChestDataset(Dataset):
    def __init__(self, df, img_dir, transform=None, is_test=False):
        self.df = df
        self.img_dir = img_dir
        self.transform = transform
        self.is_test = is_test

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        # Extract Image ID and handle file extension [cite: 121]
        name = self.df.iloc[idx, 0]
        img_filename = name if name.endswith('.png') else f"{name}.png"
        path = os.path.join(self.img_dir, img_filename)
        
        # Load Image and convert to RGB [cite: 12]
        img = Image.open(path).convert('RGB')
        if self.transform:
            img = self.transform(img)
        
        # Safely handle labels for training vs inference [cite: 103, 105]
        if self.is_test:
            # Test set has no labels, return dummy value to avoid argmax error
            return img, torch.tensor(0), name
        
        # Multi-class classification: exactly one pathology per image [cite: 15, 113, 148]
        label = torch.tensor(np.argmax(self.df.iloc[idx, 1:].values), dtype=torch.long)
        return img, label, name

# 5. Model: ResNet34 Architecture [cite: 151]
model = models.resnet34(weights='IMAGENET1K_V1')
model.fc = nn.Linear(model.fc.in_features, 20) # 20 Pathology classes [cite: 7, 100]
model = model.to('cuda')

# Loss function incorporates class weights to handle imbalance [cite: 39, 46, 84]
criterion = nn.CrossEntropyLoss(weight=class_weights, label_smoothing=0.1)
optimizer = optim.AdamW(model.parameters(), lr=config.lr)

# 6. Training Setup with Gradient Accumulation [cite: 159, 162]
train_data, val_data = train_test_split(train_df, test_size=0.12, random_state=42)
transform = transforms.Compose([
    transforms.Resize((config.img_size, config.img_size)),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
])

train_loader = DataLoader(ChestDataset(train_data, IMG_DIR, transform), 
                          batch_size=config.batch_size, shuffle=True, num_workers=2)

accumulation_steps = 4 # Effective batch size = 8 * 4 = 32

print(f"Starting training for {config.epochs} epochs...")

for epoch in range(config.epochs):
    model.train()
    t_loss = 0
    optimizer.zero_grad()
    
    for i, (imgs, labels, _) in enumerate(tqdm(train_loader, desc=f"Epoch {epoch+1}")):
        imgs, labels = imgs.to('cuda'), labels.to('cuda')
        
        outputs = model(imgs)
        loss = criterion(outputs, labels)
        loss = loss / accumulation_steps
        loss.backward()
        
        if (i + 1) % accumulation_steps == 0:
            optimizer.step()
            optimizer.zero_grad()
            
        t_loss += loss.item() * accumulation_steps
    
    # Memory Management for Kaggle Environment
    gc.collect()
    torch.cuda.empty_cache()
    
    avg_loss = t_loss / len(train_loader)
    wandb.log({"epoch": epoch+1, "loss": avg_loss})
    print(f"Epoch {epoch+1} | Loss: {avg_loss:.4f}")

# 7. Final Submission Generation [cite: 144, 145]
model.eval()
results = []
test_df = pd.read_csv(os.path.join(BASE_PATH, "test.csv"))
# Set is_test=True to use the safe label handling [cite: 105]
test_loader = DataLoader(ChestDataset(test_df, IMG_DIR, transform, is_test=True), batch_size=config.batch_size)

print("Generating final predictions for Kaggle...")
with torch.no_grad():
    for imgs, _, ids in tqdm(test_loader):
        imgs = imgs.to('cuda')
        outputs = model(imgs)
        preds = torch.argmax(outputs, dim=1).cpu().numpy()
        
        for i, p in enumerate(preds):
            # One-hot encoding: exactly one "1" per row [cite: 107, 148, 149]
            row = [0] * 20
            row[p] = 1
            results.append([ids[i]] + row)

# Save to submission.csv for Kaggle evaluation [cite: 145, 146]
pd.DataFrame(results, columns=['id'] + class_names).to_csv("submission.csv", index=False)
print("Submission file saved: submission.csv")
